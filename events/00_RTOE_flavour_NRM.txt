namespace = RTOE_flavour_NRM

#Administration of Ireland
country_event = {
	id = RTOE_flavour_NRM.1
	title = RTOE_flavour_NRM.1.t
	desc = RTOE_flavour_NRM.1.d
	picture = REFORM_eventPicture
	
	fire_only_once = yes
	is_triggered_only = yes

	option = { #Establish them as the chief authority over Ireland
		name = RTOE_flavour_NRM.1.a
		ai_chance = { factor = 100 }
		if = {
			limit = {
				NOT = { exists = DUB }
			}
			373 = {
				add_core = DUB
			}
			release = DUB
		}
		create_subject = {
			subject_type = feudal_vassal
			who = DUB
		}
		every_owned_province = {
			limit = {
				OR = {
					area = leinster_area
					area = munster_area
					area = connacht_area
					area = meath_area
					area = west_meath_area
					area = ulster_area
					area = west_ulster_area
				}
			}
			cede_province = DUB
		}
		DUB = {
			hidden_effect = {
				change_country_color = {
					color = { 210 45 45 }
				}
			}
			override_country_name = IRE
		}
		every_province = {
			limit = {
				OR = {
					area = leinster_area
					area = munster_area
					area = connacht_area
					area = meath_area
					area = west_meath_area
					area = ulster_area
					area = west_ulster_area
				}
			}
			add_permanent_claim = DUB
		}
	}
	
	option = { #Let's not give them too much power
		name = RTOE_flavour_NRM.1.b
		ai_chance = { factor = 0 }
		add_adm_power = 100
	}
}

#Harrying of the North
country_event = {
	id = RTOE_flavour_NRM.2
	title = RTOE_flavour_NRM.2.t
	desc = RTOE_flavour_NRM.2.d
	picture = IMPALED_SOLDIERS_eventPicture

	fire_only_once = yes
	is_triggered_only = yes

	option = { #Burn it!
		name = RTOE_flavour_NRM.2.a
		every_owned_province = {
			limit = {
				is_core = NOL
			}
			remove_core = NOL
		}

		every_owned_province = {
			limit = {
				culture = northumbrian
			}
			add_devastation = 100
			change_culture = english
			change_religion = ROOT
			add_base_production = -2
			add_base_manpower = -2
			add_base_tax = -2
		}
	}

	option = { #Try to integrate it
		name = RTOE_flavour_NRM.2.b
		add_adm_power = 200
		if = {
			limit = {
				NOT = { accepted_culture = northumbrian }
			}
			add_country_modifier = {
				name = NRM_rule_normandia
				duration = -1
			}
			custom_tooltip = RTOE_flavour_NRM.2.tooltip
			add_accepted_culture = northumbrian
		}
	}
}
